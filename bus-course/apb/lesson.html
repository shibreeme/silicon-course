APB：让一次写入只生效一次

目标是给软件提供一组可预测的寄存器。沿一条“写配置，再读回”的路径，先确定事务完成的时钟沿，再决定哪个 always_ff 可以修改状态。

原理与演进端口连接概念自测RTL 实验

为什么需要两个阶段？

简单片上外设很少连续传输大块数据。APB 给地址译码留出一个明确的建立周期，之后才进入访问阶段。上游的处理器或 AXI 桥充当请求端，下游的定时器、UART 或加速器控制寄存器充当外设。两个端点共享 PCLK。

早期 APB 面向简单外设。APB3 引入 PREADY 和 PSLVERR，使外设能延长访问并报告失败；APB4 增加 PSTRB 字节选通和 PPROT 保护属性。下方实验采用 APB3 的 32 位整字访问约定。

  1. 建立阶段：请求端置 PSEL，令 PENABLE 为 0，提供地址和读写方向。若写入，此时给出数据。
  2. 访问阶段：下一拍置 PENABLE。外设可拉低 PREADY 插入等待；请求端保持地址、方向和写数据。
  3. 完成边沿：在 PSEL、PENABLE、PREADY 同为 1 的上升沿采样读数据和错误信号。下一笔即使访问同一外设，也要重新经历 PENABLE 为 0 的建立阶段。

                                      │ PREADY, PRDATA, PSLVERR

信号与位宽驱动端 → 采样端连接和检查要点
PCLK、PRESETn 各 1 位系统 → 两个模块代码名 clk、rst_n。共同上升沿，低有效复位。
PSEL、PENABLE、PWRITE 各 1 位请求端 → 外设多外设系统按地址译码产生每个外设的 PSEL。单独观察 PENABLE 无法确定谁被访问。
PADDR[7:0]、PWDATA[31:0]请求端 → 外设实验地址按字节编码。请求端先锁存命令，等待期间地址和写值保持稳定。
PREADY、PSLVERR、PRDATA[31:0]外设 → 请求端完成沿读取结果。扩展为多个外设时，按被选外设复用返回通路。

先回答，再核对波形

等待持续三拍，写计数器应加几次？

成功完成时加一次。若用 PSEL && PWRITE 作写使能，建立和等待阶段都会错误触发写入。

请求端能看到 PREADY=1 就提前结束建立阶段吗？

建立阶段固定占一拍。PREADY 只在双方进入访问阶段后参与完成判定。零等待传输也至少有建立和访问两个时钟周期。

两笔写能共享一个连续为高的 PENABLE 吗？

连续访问同一设备可让 PSEL 保持高，但两个访问阶段之间需要一个新的建立阶段。本实验请求端还回到 IDLE 接收下一条命令。

写只读地址返回错误，原寄存器是否改变？

本寄存器组约定失败写无副作用，测试会读回计数器验证。设计其他外设时须明确错误与副作用的契约。

端到端实验：写入、等待、读回、拒绝

0x00 是 32 位可读写寄存器，复位值为 0；0x04 是成功写入次数，只读。外设为每次传输插入一拍等待。向只读地址写入或访问未映射地址返回错误。命令输入侧的 valid/ready 握手与 APB 的两阶段操作分属不同接口。

扩展练习：把等待长度变为可配置的 0 至 3 拍，再用相同的读写序列验证。此时提交次数不变，等待总数应随配置改变。进入 AXI 课程时，找出地址、数据和响应各自的提交边沿。

apb.sv
module apb_requester(
    input logic clk, rst_n, cmd_valid, cmd_write,
    output logic cmd_ready, done, error,
    input logic [7:0] cmd_addr,
    input logic [31:0] cmd_data,
    output logic [31:0] read_data,
    output logic psel, penable, pwrite,
    output logic [7:0] paddr,
    output logic [31:0] pwdata,
    input logic pready, pslverr,
    input logic [31:0] prdata
);
    typedef enum logic [1:0] {IDLE, SETUP, ACCESS} state_t;
    state_t state;
    assign cmd_ready = rst_n && state == IDLE;
    assign psel = state != IDLE;
    assign penable = state == ACCESS;
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            {done,error,pwrite,paddr,pwdata,read_data} <= '0;
        end else begin
            done <= 0;
            case (state)
                IDLE: if (cmd_valid) begin
                    pwrite <= cmd_write;
                    paddr <= cmd_addr;
                    pwdata <= cmd_data;
                    state <= SETUP;
                end
                SETUP: state <= ACCESS;
                ACCESS: if (pready) begin
                    done <= 1;
                    error <= pslverr;
                    if (!pwrite) read_data <= prdata;
                    state <= IDLE;
                end
                default: state <= IDLE;
            endcase
        end
    end
endmodule

module apb_registers(
    input logic clk, rst_n, psel, penable, pwrite,
    input logic [7:0] paddr,
    input logic [31:0] pwdata,
    output logic pready, pslverr,
    output logic [31:0] prdata
);
    // Address 0 is scratch; address 4 is a read-only write counter.
    logic [31:0] scratch, writes;
    logic waiting;
    assign pready = psel && penable && !waiting;
    assign pslverr = pready && !(paddr == 0 || (paddr == 4 && !pwrite));
    assign prdata = (paddr == 0) ? scratch : (paddr == 4) ? writes : 0;
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            scratch <= 0;
            writes <= 0;
            waiting <= 0;
        end else begin
            waiting <= psel && !penable;
            if (pready && pwrite && !pslverr) begin
                scratch <= pwdata;
                writes <= writes + 1;
            end
        end
    end
endmodule
tb_apb.sv
module tb_apb;
    bit clk;
    always #5 clk = ~clk;
    logic rst_n, cmd_valid, cmd_write, cmd_ready, done, error;
    logic [7:0] cmd_addr, paddr;
    logic [31:0] cmd_data, read_data, pwdata, prdata;
    logic psel, penable, pwrite, pready, pslverr;
    apb_requester u_req(.*);
    apb_registers u_regs(.*);
    int setups, stalls, transfers;
    logic [40:0] snapshot;
    always @(posedge clk) if (rst_n) begin
        if (psel && !penable) begin
            setups++;
            snapshot = {paddr,pwrite,pwdata};
        end
        if (penable) begin
            if (!psel || {paddr,pwrite,pwdata} !== snapshot) $fatal(1,"unstable access");
            if (pready) transfers++; else stalls++;
        end
    end
    task transact(input bit wr, input logic [7:0] addr,
                  input logic [31:0] data, expected, input bit fault);
        @(negedge clk);
        if (!cmd_ready) $fatal(1,"requester busy");
        cmd_valid=1; cmd_addr=addr; cmd_write=wr; cmd_data=data;
        @(negedge clk);
        cmd_valid=0; cmd_data=~data; cmd_addr=~addr;
        while (!done) @(negedge clk);
        if (error !== fault || (!wr && read_data !== expected))
            $fatal(1,"bad response addr=%h",addr);
    endtask
    initial begin
        rst_n=0; cmd_valid=0;
        repeat (2) @(negedge clk);
        rst_n=1;
        transact(0,0,0,0,0);
        transact(1,0,32'hcafe1234,0,0);
        transact(1,4,32'hdeadbeef,0,1);
        transact(0,8,0,0,1);
        transact(0,0,0,32'hcafe1234,0);
        transact(0,4,0,1,0);
        if (setups!=6 || stalls!=6 || transfers!=6) $fatal(1,"phase counts");
        $display("PASS APB: 6 transfers, 6 wait cycles, error and side effects checked");
        $finish;
    end
    initial #2000 $fatal(1,"timeout");
endmodule
